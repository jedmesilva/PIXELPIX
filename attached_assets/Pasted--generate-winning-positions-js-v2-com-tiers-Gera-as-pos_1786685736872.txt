// generate-winning-positions.js (v2 — com tiers)
//
// Gera as posições premiadas (winning_positions), agora atribuindo também a
// DENOMINAÇÃO (tier) de cada uma, de forma segura e eficiente:
//   1. Sorteia N ids únicos no intervalo [0, TOTAL_CELLS) — sem precisar embaralhar
//      o array de 1 milhão inteiro (reservoir sampling via Set, O(N) na prática).
//   2. Gera a lista de tier_id respeitando as QUANTIDADES EXATAS por denominação
//      (não um sorteio livre por peso — isso garante que o total de posições por
//      tier bate exatamente com o planejado) e embaralha essa lista.
//   3. Associa cada id sorteado a um tier_id da lista embaralhada.
//   4. Calcula o hash SHA-256 do commit (lista ordenada de pares cell_id/tier_id)
//      ANTES de inserir qualquer linha — esse hash deve ser publicado publicamente
//      antes de abrir a venda.
//   5. Insere em lote (multi-row INSERT em chunks), muito mais rápido que inserir
//      célula por célula.
//
// Rodar UMA VEZ, antes de vender a primeira célula. Depois de rodado, não deve
// ser rodado de novo para o mesmo lote (as tabelas geradas aqui são imutáveis).

const crypto = require('crypto');
const { Pool } = require('pg');

const TOTAL_CELLS = 1_000_000;

// Teto do pool de prêmios: R$1.000.000,00 FIXO, como decisão de negócio —
// independente de quanto o caixa efetivamente arrecada.
//
// Importante: NÃO calcule isso como `TOTAL_CELLS * preço_da_célula`. Com o
// sistema de precificação progressiva (célula começa em R$1 e sobe até um teto
// de R$1.000 conforme o pool de células disponíveis diminui), o caixa real
// arrecadado é muito maior e incerto — em simulação, na casa de R$35M+. O
// tamanho do pool de prêmios é uma escolha de negócio independente disso, não
// uma fração fixa da receita.
const MAX_POOL_VALUE_CENTS = 100_000_000; // R$ 1.000.000,00 — ajuste aqui
                                            // diretamente se o valor do pool mudar,
                                            // nunca derivando de TOTAL_CELLS × preço
const CHUNK_SIZE = 5_000; // linhas por INSERT — evita um único INSERT gigante
                           // e mantém memória/params dentro de limites confortáveis

// Definição dos tiers (denominações) — ajuste livremente aqui.
// nominal_value_cents = valor MÉDIO esperado por posição do tier (não é um valor
// individual travado — ver seção 8.2 do doc para o porquê).
// A soma de (quantidade * nominal_value_cents) de TODOS os tiers precisa fechar
// EXATAMENTE R$1.000.000 (o teto: nunca pode passar do valor total arrecadável
// pelo pool de 1.000.000 células) — os números abaixo já fecham nesse total.
const TIERS = [
  { id: 1, label: 'R$10',     quantidade: 40_000, nominal_value_cents: 1_000 },      // R$400.000
  { id: 2, label: 'R$100',    quantidade: 3_000,  nominal_value_cents: 10_000 },     // R$300.000
  { id: 3, label: 'R$1.000',  quantidade: 150,    nominal_value_cents: 100_000 },    // R$150.000
  { id: 4, label: 'R$10.000', quantidade: 10,     nominal_value_cents: 1_000_000 },  // R$100.000
  { id: 5, label: 'R$50.000', quantidade: 1,      nominal_value_cents: 5_000_000 },  // R$50.000
];
// total: 43.161 posições premiadas, R$1.000.000 exatos em prêmios (400k + 300k +
// 150k + 100k + 50k). Ajuste as quantidades como quiser, mas sempre confira que a
// soma final bate em R$1.000.000 — o script valida isso abaixo antes de gravar
// qualquer coisa.

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
});

// Sorteia N ids únicos de [0, total) sem gerar o array inteiro de 1M em memória
// para embaralhar — usa um Set e vai sorteando até ter N ids distintos.
// Para N << total, a taxa de colisão é baixa o suficiente para isso ser rápido
// (esperança de tentativas ≈ total * -ln(1 - N/total)).
function sortearIdsUnicos(total, n) {
  const escolhidos = new Set();
  while (escolhidos.size < n) {
    const id = crypto.randomInt(0, total); // randomInt é criptograficamente seguro
    escolhidos.add(id);
  }
  return Array.from(escolhidos);
}

// Gera a lista de tier_id respeitando as quantidades EXATAS por tier (não um
// sorteio livre por peso), depois embaralha essa lista com Fisher-Yates — assim
// a atribuição de tier para cada posição fica imprevisível, mas a quantidade
// total por tier é garantida desde o início.
function gerarAtribuicaoDeTiers(tiers) {
  const atribuicoes = [];
  for (const tier of tiers) {
    for (let i = 0; i < tier.quantidade; i++) {
      atribuicoes.push(tier.id);
    }
  }
  for (let i = atribuicoes.length - 1; i > 0; i--) {
    const j = crypto.randomInt(0, i + 1);
    [atribuicoes[i], atribuicoes[j]] = [atribuicoes[j], atribuicoes[i]];
  }
  return atribuicoes;
}

function calcularCommitHash(paresOrdenados) {
  // pares no formato "cellId:tierId", ordenados por cellId — determinístico
  const conteudo = paresOrdenados.map((p) => `${p.cellId}:${p.tierId}`).join(',');
  return crypto.createHash('sha256').update(conteudo).digest('hex');
}

async function inserirWinningPositionsEmLotes(client, atribuicoesFinais) {
  for (let i = 0; i < atribuicoesFinais.length; i += CHUNK_SIZE) {
    const lote = atribuicoesFinais.slice(i, i + CHUNK_SIZE);

    const valoresSql = [];
    const params = [];
    lote.forEach((item, idx) => {
      const base = idx * 2;
      valoresSql.push(`($${base + 1}, $${base + 2})`);
      params.push(item.cellId, item.tierId);
    });

    await client.query(
      `INSERT INTO winning_positions (cell_id, tier_id) VALUES ${valoresSql.join(',')}`,
      params
    );

    console.log(
      `Inseridas ${Math.min(i + CHUNK_SIZE, atribuicoesFinais.length)} / ${atribuicoesFinais.length}`
    );
  }
}

async function main() {
  const client = await pool.connect();
  try {
    // trava simples: se prize_tier_batch já existe, aborta — evita rodar duas
    // vezes por engano e gerar um segundo commit para o mesmo lote
    const existente = await client.query(`SELECT id FROM prize_tier_batch WHERE id = 1`);
    if (existente.rows.length > 0) {
      throw new Error(
        'prize_tier_batch já existe — este script só deve rodar uma vez, antes da 1ª venda.'
      );
    }

    const totalWinningPositions = TIERS.reduce((sum, t) => sum + t.quantidade, 0);
    const totalPoolCents = TIERS.reduce(
      (sum, t) => sum + t.quantidade * t.nominal_value_cents,
      0
    );

    // validação dura: aborta ANTES de sortear ou gravar qualquer coisa se o total
    // planejado ultrapassar o teto de R$1.000.000 — nunca pode passar do valor
    // máximo arrecadável mesmo que as 1.000.000 células sejam todas vendidas
    if (totalPoolCents > MAX_POOL_VALUE_CENTS) {
      throw new Error(
        `Pool de prêmios (R$ ${(totalPoolCents / 100).toFixed(2)}) ultrapassa o teto ` +
        `de R$ ${(MAX_POOL_VALUE_CENTS / 100).toFixed(2)}. Ajuste as quantidades/valores em TIERS.`
      );
    }
    if (totalWinningPositions > TOTAL_CELLS) {
      throw new Error(
        `Total de posições premiadas (${totalWinningPositions}) ultrapassa o total ` +
        `de células disponíveis (${TOTAL_CELLS}).`
      );
    }

    console.log(`Sorteando ${totalWinningPositions} posições únicas entre ${TOTAL_CELLS}...`);
    const ids = sortearIdsUnicos(TOTAL_CELLS, totalWinningPositions);

    console.log('Atribuindo tiers às posições sorteadas...');
    const tiersEmbaralhados = gerarAtribuicaoDeTiers(TIERS);

    // associa cada id sorteado a um tier_id (mesma ordem de índice, ambos já
    // "aleatórios" e independentes entre si)
    const atribuicoesFinais = ids.map((cellId, idx) => ({
      cellId,
      tierId: tiersEmbaralhados[idx],
    }));

    const paresOrdenados = [...atribuicoesFinais].sort((a, b) => a.cellId - b.cellId);
    const commitHash = calcularCommitHash(paresOrdenados);
    console.log(`Commit hash (publicar isso antes de abrir a venda): ${commitHash}`);

    await client.query('BEGIN');

    // popula prize_pool — uma linha por tier
    for (const tier of TIERS) {
      const totalValueCents = tier.quantidade * tier.nominal_value_cents;
      await client.query(
        `INSERT INTO prize_pool
           (tier_id, label, nominal_value_cents, total_value_cents,
            total_positions, remaining_value_cents, remaining_positions)
         VALUES ($1, $2, $3, $4, $5, $4, $5)`,
        [tier.id, tier.label, tier.nominal_value_cents, totalValueCents, tier.quantidade]
      );
    }

    await inserirWinningPositionsEmLotes(client, atribuicoesFinais);

    await client.query(
      `INSERT INTO prize_tier_batch (id, commit_hash) VALUES (1, $1)`,
      [commitHash]
    );

    await client.query('COMMIT');

    console.log('Concluído. winning_positions, prize_pool e prize_tier_batch populados.');
    console.log(`Total de posições premiadas: ${totalWinningPositions}`);
    console.log(`Total do pool de prêmios: R$ ${(totalPoolCents / 100).toFixed(2)}`);
    console.log(`Guarde o commit hash publicado: ${commitHash}`);
  } catch (err) {
    await client.query('ROLLBACK');
    console.error('Erro — nada foi gravado:', err.message);
    process.exitCode = 1;
  } finally {
    client.release();
    await pool.end();
  }
}

main();
